def extract(p,outp):
    doc=Document(p)
    def it(parent):
        for ch in parent.element.body.iterchildren():
            if ch.tag==qn('w:p'): yield Paragraph(ch,parent)
            elif ch.tag==qn('w:tbl'): yield Table(ch,parent)
    out=[];tc=0

    io.open(outp,"w",encoding="utf-8").write("\n".join(out))
